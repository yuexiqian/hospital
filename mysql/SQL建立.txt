CREATE TABLE users (
    user_id     BIGINT AUTO_INCREMENT PRIMARY KEY,
    login_name  VARCHAR(50) NOT NULL UNIQUE COMMENT '登录账号：患者为手机号，医护为工号等',
    password    VARCHAR(100) NOT NULL COMMENT '加密后的密码',
    phone       VARCHAR(20) DEFAULT NULL COMMENT '联系电话',
    role        VARCHAR(20) NOT NULL COMMENT 'PATIENT / DOCTOR / NURSE / PHARMACIST / ADMIN',
    status      TINYINT DEFAULT 1 COMMENT '1=正常,0=停用',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE patients (
    patient_id  BIGINT AUTO_INCREMENT PRIMARY KEY,
    name        VARCHAR(50) NOT NULL COMMENT '就诊人姓名',
    id_type     VARCHAR(20) NOT NULL COMMENT '证件类型：身份证/护照/医保卡',
    id_card     VARCHAR(50) NOT NULL COMMENT '证件号码',
    user_id     BIGINT NOT NULL COMMENT '关联用户ID',
    phone       VARCHAR(20) DEFAULT NULL COMMENT '联系电话',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE departments (
    dept_id     BIGINT AUTO_INCREMENT PRIMARY KEY,
    dept_name   VARCHAR(50) NOT NULL UNIQUE COMMENT '科室名称',
    description VARCHAR(200) DEFAULT NULL COMMENT '科室描述',
    location    VARCHAR(100) DEFAULT NULL COMMENT '科室位置',
    director    VARCHAR(50) DEFAULT NULL COMMENT '科室负责人'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE doctors (
    doctor_id   BIGINT AUTO_INCREMENT PRIMARY KEY,
    work_no     VARCHAR(50) NOT NULL UNIQUE COMMENT '医生工号',
    name        VARCHAR(50) NOT NULL COMMENT '医生姓名',
    title       VARCHAR(50) DEFAULT NULL COMMENT '职称：主任医师/副主任医师等',
    dept_id     BIGINT NOT NULL COMMENT '所属科室ID',
    specialty   VARCHAR(200) DEFAULT NULL COMMENT '擅长领域',
    FOREIGN KEY (dept_id) REFERENCES departments(dept_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE nurses (
    nurse_id    BIGINT AUTO_INCREMENT PRIMARY KEY,
    work_no     VARCHAR(50) NOT NULL UNIQUE COMMENT '护士工号',
    name        VARCHAR(50) NOT NULL COMMENT '护士姓名',
    dept_id     BIGINT NOT NULL COMMENT '所属科室ID',
    FOREIGN KEY (dept_id) REFERENCES departments(dept_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE pharmacists (
    pharmacist_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    work_no      VARCHAR(50) NOT NULL UNIQUE COMMENT '药师工号',
    name         VARCHAR(50) NOT NULL COMMENT '药师姓名',
    pharmacy     VARCHAR(50) NOT NULL COMMENT '所属药房'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE medicines (
    medicine_id  BIGINT AUTO_INCREMENT PRIMARY KEY,
    name         VARCHAR(100) NOT NULL COMMENT '药品名称',
    spec         VARCHAR(50) NOT NULL COMMENT '药品规格',
    price        DECIMAL(10,2) NOT NULL COMMENT '单价',
    stock        INT NOT NULL DEFAULT 0 COMMENT '库存数量',
    indications  VARCHAR(500) DEFAULT NULL COMMENT '适应症',
    contraindications VARCHAR(500) DEFAULT NULL COMMENT '禁忌症'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE medicine_knowledge (
    knowledge_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    medicine_id  BIGINT NOT NULL UNIQUE COMMENT '关联药品ID',
    usage_guide  VARCHAR(1000) DEFAULT NULL COMMENT '用药指导',
    side_effects VARCHAR(1000) DEFAULT NULL COMMENT '副作用',
    interactions VARCHAR(1000) DEFAULT NULL COMMENT '药物相互作用',
    FOREIGN KEY (medicine_id) REFERENCES medicines(medicine_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE registration_records (
    reg_id      BIGINT AUTO_INCREMENT PRIMARY KEY,
    patient_id  BIGINT NOT NULL COMMENT '就诊人ID',
    dept_id     BIGINT NOT NULL COMMENT '就诊科室ID',
    doctor_id   BIGINT NOT NULL COMMENT '接诊医生ID',
    reg_time    DATETIME NOT NULL COMMENT '挂号时间',
    status      VARCHAR(20) NOT NULL COMMENT '状态：已完成/待就诊/已取消',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (patient_id) REFERENCES patients(patient_id),
    FOREIGN KEY (dept_id) REFERENCES departments(dept_id),
    FOREIGN KEY (doctor_id) REFERENCES doctors(doctor_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE triage_records (
    triage_id   BIGINT AUTO_INCREMENT PRIMARY KEY,
    reg_id      BIGINT NOT NULL UNIQUE COMMENT '关联挂号ID',
    symptoms    VARCHAR(1000) NOT NULL COMMENT '症状描述',
    recommend_dept_id BIGINT NOT NULL COMMENT '推荐科室ID',
    triage_time DATETIME NOT NULL COMMENT '分诊时间',
    is_manual   TINYINT DEFAULT 0 COMMENT '是否人工辅助：1=是,0=否',
    nurse_id    BIGINT DEFAULT NULL COMMENT '复核护士ID',
    FOREIGN KEY (reg_id) REFERENCES registration_records(reg_id),
    FOREIGN KEY (recommend_dept_id) REFERENCES departments(dept_id),
    FOREIGN KEY (nurse_id) REFERENCES nurses(nurse_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE prescriptions (
    presc_id    BIGINT AUTO_INCREMENT PRIMARY KEY,
    reg_id      BIGINT NOT NULL COMMENT '关联挂号ID',
    medicine_id BIGINT NOT NULL COMMENT '药品ID',
    quantity    INT NOT NULL COMMENT '药品数量',
    doctor_id   BIGINT NOT NULL COMMENT '开方医生ID',
    presc_time  DATETIME NOT NULL COMMENT '开方时间',
    FOREIGN KEY (reg_id) REFERENCES registration_records(reg_id),
    FOREIGN KEY (medicine_id) REFERENCES medicines(medicine_id),
    FOREIGN KEY (doctor_id) REFERENCES doctors(doctor_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE dispense_records (
    dispense_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    presc_id    BIGINT NOT NULL UNIQUE COMMENT '关联处方ID',
    pharmacist_id BIGINT NOT NULL COMMENT '发药药师ID',
    dispense_time DATETIME NOT NULL COMMENT '发药时间',
    quantity    INT NOT NULL COMMENT '发药数量',
    FOREIGN KEY (presc_id) REFERENCES prescriptions(presc_id),
    FOREIGN KEY (pharmacist_id) REFERENCES pharmacists(pharmacist_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE medicine_pharmacist_relation (
    medicine_id  BIGINT NOT NULL COMMENT '药品ID',
    pharmacist_id BIGINT NOT NULL COMMENT '药师ID',
    PRIMARY KEY (medicine_id, pharmacist_id),
    FOREIGN KEY (medicine_id) REFERENCES medicines(medicine_id),
    FOREIGN KEY (pharmacist_id) REFERENCES pharmacists(pharmacist_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
